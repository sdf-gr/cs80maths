cmake_minimum_required(VERSION 3.25)
project(cs80maths VERSION 0.3.0 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This CMake build is for Windows VST3. Use scripts/build.sh for macOS AUv3.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
if(NOT VST3_SDK_ROOT)
 message(FATAL_ERROR "Set VST3_SDK_ROOT to the pinned VST3 SDK checkout; see docs/WINDOWS.md")
endif()
add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/sdk")
smtg_add_vst3plugin(cs80maths Windows/Plugin.hpp Windows/Processor.cpp Windows/Controller.cpp Windows/Editor.cpp Windows/Factory.cpp CS80/DSP/Engine.cpp)
target_include_directories(cs80maths PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(cs80maths PRIVATE sdk comctl32 gdi32 user32)
target_compile_definitions(cs80maths PRIVATE NOMINMAX UNICODE _UNICODE)
smtg_target_configure_version_file(cs80maths)
add_executable(cs80maths_tests "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp" Windows/Tests.cpp Windows/Processor.cpp Windows/Controller.cpp Windows/Editor.cpp CS80/DSP/Engine.cpp)
target_include_directories(cs80maths_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(cs80maths_tests PRIVATE sdk sdk_hosting comctl32 gdi32 user32)
target_compile_definitions(cs80maths_tests PRIVATE NOMINMAX UNICODE _UNICODE)
enable_testing()
add_test(NAME cs80maths_windows COMMAND cs80maths_tests)

if(MSVC)
 target_compile_options(cs80maths PRIVATE /utf-8)
 target_compile_options(cs80maths_tests PRIVATE /utf-8)
endif()
